cmake_minimum_required(VERSION 3.18)
project(lodos_fg C)
set(CMAKE_C_STANDARD 11)
find_package(Vulkan REQUIRED)
find_package(PNG REQUIRED)
find_program(GLSLANG_VALIDATOR glslangValidator REQUIRED)
set(SHADER_OUT ${CMAKE_CURRENT_BINARY_DIR}/shaders)
file(MAKE_DIRECTORY ${SHADER_OUT})
foreach(shader easu rcas)
 add_custom_command(OUTPUT ${SHADER_OUT}/${shader}.spv
  COMMAND ${GLSLANG_VALIDATOR} -V --target-env vulkan1.1 -I${CMAKE_CURRENT_SOURCE_DIR}/third_party/amd-fsr1 -o ${SHADER_OUT}/${shader}.spv ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}.comp
  DEPENDS shaders/${shader}.comp third_party/amd-fsr1/ffx_a.h third_party/amd-fsr1/ffx_fsr1.h VERBATIM)
 list(APPEND SPV ${SHADER_OUT}/${shader}.spv)
endforeach()
add_custom_target(fsr1_shaders DEPENDS ${SPV})
add_library(lodos-fg STATIC src/fsr1_core.c)
add_dependencies(lodos-fg fsr1_shaders)
target_include_directories(lodos-fg PUBLIC include PRIVATE third_party/amd-fsr1)
target_link_libraries(lodos-fg PUBLIC Vulkan::Vulkan)
add_executable(fsr1_offscreen test/fsr1_offscreen.c)
add_dependencies(fsr1_offscreen fsr1_shaders)
target_compile_definitions(fsr1_offscreen PRIVATE FSR_EASU_SPV="${SHADER_OUT}/easu.spv" FSR_RCAS_SPV="${SHADER_OUT}/rcas.spv" OUTPUT_PNG="${CMAKE_SOURCE_DIR}/../../docs/screens/T068-fsr1.png")
target_link_libraries(fsr1_offscreen PRIVATE lodos-fg PNG::PNG m)
